cmake_minimum_required(VERSION 3.21)
project(Practical_03_GroundStation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOUIC_OPTIONS --no-autoconnection)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Charts Test)
set(MODEL src/uav.cpp src/uav.h src/sensor.cpp src/sensor.h src/navigation.cpp src/navigation.h src/controller.cpp src/controller.h src/autopilot.cpp src/autopilot.h)
add_library(station_model STATIC ${MODEL})
target_include_directories(station_model PUBLIC src)
target_link_libraries(station_model PUBLIC Qt6::Core)
set(VIEW src/mainwindow.cpp src/mainwindow.h src/mainwindow.ui src/trajectoryview.cpp src/trajectoryview.h resources.qrc)
add_executable(GroundStation MACOSX_BUNDLE src/main.cpp ${VIEW})
target_link_libraries(GroundStation PRIVATE station_model Qt6::Widgets Qt6::Charts)
enable_testing()
add_executable(ModelTests tests/model_tests.cpp)
target_link_libraries(ModelTests PRIVATE station_model Qt6::Test)
add_test(NAME model COMMAND ModelTests)
add_executable(GuiTests tests/gui_tests.cpp ${VIEW})
target_link_libraries(GuiTests PRIVATE station_model Qt6::Test Qt6::Widgets Qt6::Charts)
add_test(NAME gui COMMAND GuiTests)
set_tests_properties(gui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;STATION_VERIFY_DIR=${CMAKE_BINARY_DIR}/verification")
set_target_properties(GroundStation PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Наземная станция")
